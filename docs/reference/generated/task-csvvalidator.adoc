[cols="4,6,1", options="header", role="gretl-task-properties"]
|===
| Feld | Beschreibung | Erforderlich

| [.dsl-signature]#*dataFiles* = list#
| Zu validierende [.acronym]#CSV#-Dateien +
Beispiel: [.code-fragment]#`dataFiles = [file('data/input')]`#
| [.required]#ja#

| [.dsl-signature]#*modelNames* = list#
| Liste der [.acronym]#INTERLIS#-Modellnamen; die passenden Modelle müssen über modelDirectories auffindbar sein. +
Vorgabe: [.code-fragment]#`[]`# +
Beispiel: [.code-fragment]#`modelNames = ['Model']`#
| [.optional]#nein#

| [.dsl-signature]#*modelDirectories* = list#
| Liste lokaler Modellverzeichnisse als Path oder Repository-Adressen als Text; unterstützt auch IOX-Platzhalter wie %[.acronym]#XTF#_DIR. +
Vorgabe: [.code-fragment]#`[]`# +
Beispiel: [.code-fragment]#`modelDirectories = [file('models')]`#
| [.optional]#nein#

| [.dsl-signature]#*configFile* = path#
| Lokale Validierungskonfiguration mit zusätzlichen Regeln oder bewusst deaktivierten Prüfungen. +
Beispiel: [.code-fragment]#`configFile = file('validation.toml')`#
| [.optional]#nein#

| [.dsl-signature]#*configRepositoryId* = text#
| Repository-ID der Validierungskonfiguration; Alternative zur lokalen configFile. +
Beispiel: [.code-fragment]#`configRepositoryId = 'ch.example.validation'`#
| [.optional]#nein#

| [.dsl-signature]#*metaConfigFile* = path#
| Lokale Meta-Konfiguration für die Validierung und deren Modell-/Regelkonfiguration. +
Beispiel: [.code-fragment]#`metaConfigFile = file('meta.toml')`#
| [.optional]#nein#

| [.dsl-signature]#*metaConfigRepositoryId* = text#
| Repository-ID einer Meta-Konfiguration; Alternative zur lokalen metaConfigFile. +
Beispiel: [.code-fragment]#`metaConfigRepositoryId = 'ch.example.meta'`#
| [.optional]#nein#

| [.dsl-signature]#*forceTypeValidation* = boolean#
| Typprüfung auch dann erzwingen, wenn sie in der Validierungskonfiguration deaktiviert wurde. +
Vorgabe: [.code-fragment]#`false`# +
Beispiel: [.code-fragment]#`forceTypeValidation = true`#
| [.optional]#nein#

| [.dsl-signature]#*disableAreaValidation* = boolean#
| Flächentopologie-Prüfung deaktivieren; andere Validierungen bleiben davon unabhängig. +
Vorgabe: [.code-fragment]#`false`# +
Beispiel: [.code-fragment]#`disableAreaValidation = true`#
| [.optional]#nein#

| [.dsl-signature]#*multiplicityOff* = boolean#
| Prüfung der Attribut- und Beziehungsmultiplizitäten deaktivieren. +
Vorgabe: [.code-fragment]#`false`# +
Beispiel: [.code-fragment]#`multiplicityOff = true`#
| [.optional]#nein#

| [.dsl-signature]#*allObjectsAccessible* = boolean#
| Dem Validator mitteilen, dass alle referenzierten Objekte im Prüfbestand verfügbar sein sollen. +
Vorgabe: [.code-fragment]#`false`# +
Beispiel: [.code-fragment]#`allObjectsAccessible = true`#
| [.optional]#nein#

| [.dsl-signature]#*skipPolygonBuilding* = boolean#
| Polygonbildung aus [.acronym]#INTERLIS#-1-Linien überspringen; keine reparierten Flächen voraussetzen. +
Vorgabe: [.code-fragment]#`false`# +
Beispiel: [.code-fragment]#`skipPolygonBuilding = true`#
| [.optional]#nein#

| [.dsl-signature]#*logFile* = path#
| Pfad der Validierungsdiagnosen; ohne expliziten Pfad verwendet der Adapter eine Datei im Arbeitsbereich. +
Beispiel: [.code-fragment]#`logFile = workPath('validation.log')`#
| [.optional]#nein#

| [.dsl-signature]#*xtfLogFile* = path#
| Zusätzliche strukturierte Fehlerdatei im [.acronym]#INTERLIS#-Modell IliVErrors. +
Beispiel: [.code-fragment]#`xtfLogFile = workPath('errors.xtf')`#
| [.optional]#nein#

| [.dsl-signature]#*proxy* = text#
| Hostname des [.acronym]#HTTP#-Proxys für Zugriffe der Fachbibliothek auf Modell- und Konfigurationsrepositories. +
Beispiel: [.code-fragment]#`proxy = 'proxy.example'`#
| [.optional]#nein#

| [.dsl-signature]#*proxyPort* = integer#
| Port des [.acronym]#HTTP#-Proxys; zusammen mit proxy verwenden. +
Beispiel: [.code-fragment]#`proxyPort = 8080`#
| [.optional]#nein#

| [.dsl-signature]#*failOnError* = boolean#
| Task bei Validierungsfehler abbrechen +
Vorgabe: [.code-fragment]#`true`# +
Beispiel: [.code-fragment]#`failOnError = false`#
| [.optional]#nein#

| [.dsl-signature]#*firstLineIsHeader* = boolean#
| Erste [.acronym]#CSV#-Zeile als Header verwenden +
Vorgabe: [.code-fragment]#`true`# +
Beispiel: [.code-fragment]#`firstLineIsHeader = false`#
| [.optional]#nein#

| [.dsl-signature]#*valueDelimiter* = text#
| Einzelnes [.acronym]#CSV#-Anführungszeichen, beispielsweise ein doppeltes Anführungszeichen. +
Beispiel: [.code-fragment]#`valueDelimiter = '"'`#
| [.optional]#nein#

| [.dsl-signature]#*valueSeparator* = text#
| Einzelnes [.acronym]#CSV#-Feldtrennzeichen, beispielsweise ein Semikolon. +
Beispiel: [.code-fragment]#`valueSeparator = ';'`#
| [.optional]#nein#

| [.dsl-signature]#*encoding* = text#
| [.acronym]#CSV#-Kodierung +
Beispiel: [.code-fragment]#`encoding = 'UTF-8'`#
| [.optional]#nein#

|===
